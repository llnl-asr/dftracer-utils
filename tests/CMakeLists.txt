# Pure-C ABI guard: compile every public C ABI header (*abi.h and c_api*.h under include/)
# with the C compiler so any C++ leak into one of them fails the build. The
# source is generated at configure time from a recursive glob so a new
# */abi.h or */c_api.h is picked up automatically without editing this file.
file(
  GLOB_RECURSE DFTRACER_UTILS_C_ABI_HEADERS
  RELATIVE ${CMAKE_SOURCE_DIR}/include
  ${CMAKE_SOURCE_DIR}/include/dftracer/*abi.h
  ${CMAKE_SOURCE_DIR}/include/dftracer/*c_api*.h)

list(SORT DFTRACER_UTILS_C_ABI_HEADERS)

set(C_ABI_GUARD_SRC ${CMAKE_CURRENT_BINARY_DIR}/generated/c_abi_guard.c)
set(C_ABI_GUARD_CONTENT
    "/* Generated by tests/CMakeLists.txt. Compiles every public C ABI header\n   (*abi.h and c_api*.h under include/) with a C compiler; a C++ leak into any of them\n   breaks this build. */\n\n"
)
foreach(hdr ${DFTRACER_UTILS_C_ABI_HEADERS})
  string(APPEND C_ABI_GUARD_CONTENT "#include <${hdr}>\n")
endforeach()
string(APPEND C_ABI_GUARD_CONTENT
       "\nint dftracer_c_abi_guard(void);\nint dftracer_c_abi_guard(void) { return 0; }\n")
file(WRITE ${C_ABI_GUARD_SRC} "${C_ABI_GUARD_CONTENT}")
set_source_files_properties(${C_ABI_GUARD_SRC} PROPERTIES LANGUAGE C)

add_library(c_abi_guard OBJECT ${C_ABI_GUARD_SRC})
target_set_warnings(c_abi_guard)
target_include_directories(
  c_abi_guard PRIVATE ${CMAKE_SOURCE_DIR}/include ${CMAKE_BINARY_DIR}/include)

add_test(NAME c_abi_guard
         COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR} --target
                 c_abi_guard --config $<CONFIG>)

# The plugin ABI manifest: the recorded shape of the headers a plugin compiles
# against (struct sizes, field offsets, enum values, integer constants and
# function pointer spellings). The generator writes a C program that asks the
# compiler for every number, so a shape change shows as one changed line.
# python3 is required and not optional: without it there is no manifest.
find_package(Python3 3.8 COMPONENTS Interpreter REQUIRED)
file(GLOB_RECURSE DFTRACER_UTILS_ABI_HEADER_DEPS
     ${CMAKE_SOURCE_DIR}/include/dftracer/*.h)
set(PLUGIN_ABI_MANIFEST_SRC
    ${CMAKE_CURRENT_BINARY_DIR}/generated/plugin_abi_manifest.c)
set(PLUGIN_ABI_MANIFEST_TXT ${CMAKE_SOURCE_DIR}/cmake/plugin_abi_manifest.txt)
set(PLUGIN_ABI_MANIFEST_WORK
    ${CMAKE_CURRENT_BINARY_DIR}/plugin_abi_manifest_work)

add_custom_command(
  OUTPUT ${PLUGIN_ABI_MANIFEST_SRC}
  COMMAND
    ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/plugin_abi_manifest.py
    --root ${CMAKE_SOURCE_DIR} --out ${PLUGIN_ABI_MANIFEST_SRC}
  DEPENDS ${DFTRACER_UTILS_ABI_HEADER_DEPS}
          ${CMAKE_SOURCE_DIR}/scripts/plugin_abi_manifest.py
  COMMENT "Writing the plugin ABI manifest source"
  VERBATIM)

add_executable(plugin_abi_manifest_gen ${PLUGIN_ABI_MANIFEST_SRC})
set_target_properties(plugin_abi_manifest_gen PROPERTIES C_STANDARD 11
                                                         OUTPUT_NAME plugin_abi_manifest)
target_set_warnings(plugin_abi_manifest_gen)
target_include_directories(
  plugin_abi_manifest_gen PRIVATE ${CMAKE_SOURCE_DIR}/include
                                  ${CMAKE_BINARY_DIR}/include)

# Records cmake/plugin_abi_manifest.txt from the headers as they are now.
add_custom_target(
  plugin_abi_manifest_update
  COMMAND
    ${CMAKE_COMMAND} -DBINARY=$<TARGET_FILE:plugin_abi_manifest_gen>
    -DMANIFEST=${PLUGIN_ABI_MANIFEST_TXT} -DREGENERATE=ON -P
    ${CMAKE_SOURCE_DIR}/cmake/scripts/check_plugin_abi_manifest.cmake
  DEPENDS plugin_abi_manifest_gen
  COMMENT "Recording cmake/plugin_abi_manifest.txt"
  VERBATIM)

foreach(test_file ${TEST_CPP_SOURCES})
  # Extract path and filename
  string(REPLACE ".cpp" "" bin_exec ${test_file})
  string(REPLACE "${CMAKE_CURRENT_SOURCE_DIR}/" "" bin_exec ${bin_exec})

  # Create valid target name (replace / with _)
  string(REPLACE "/" "_" target_name ${bin_exec})

  add_executable(${target_name} ${test_file})
  target_include_directories(${target_name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}
                                                 ${CMAKE_SOURCE_DIR}/src)
  target_link_libraries(${target_name} PRIVATE doctest::doctest testing_utilities)
  target_set_warnings(${target_name})
  target_enable_coroutine(${target_name})
  target_add_rpath(${target_name})
  
  # Pass CMAKE_BINARY_DIR to tests that need to execute binaries
  target_compile_definitions(
    ${target_name} PRIVATE CMAKE_BINARY_DIR="${CMAKE_BINARY_DIR}"
                           DFTRACER_UTILS_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")

  # Set output directory to preserve folder structure in binaries
  get_filename_component(bin_dir ${bin_exec} DIRECTORY)
  get_filename_component(bin_name ${bin_exec} NAME)

  if(bin_dir)
    set_target_properties(${target_name} PROPERTIES
      RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${bin_dir}")
  endif()

  set_target_properties(${target_name} PROPERTIES OUTPUT_NAME ${bin_name})

  # Apply coverage flags if enabled
  if(DFTRACER_UTILS_COVERAGE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target_name} PRIVATE --coverage -fprofile-arcs
                                               -ftest-coverage)
    target_link_libraries(${target_name} PRIVATE --coverage)
  endif()

  add_test(NAME ${bin_exec} COMMAND ${target_name})

  # Default timeout for all tests (safety net against hangs in CI)
  set(default_test_timeout 120)
  # A coverage build is barely optimized and counts every branch, so the heavy
  # store, index and reader tests take several times their usual time.
  if(DFTRACER_UTILS_COVERAGE)
    set(default_test_timeout 900)
  endif()
  if(DFTRACER_UTILS_ENABLE_TSAN OR DFTRACER_UTILS_ENABLE_ASAN)
    set(default_test_timeout 1800)
  endif()

  set_tests_properties(${bin_exec} PROPERTIES TIMEOUT ${default_test_timeout})

  # These thread-heavy tests contend heavily when run together on low-core CI.
  # Give them extra timeout and serialize via a shared lock.
  if(bin_exec STREQUAL "runtime/test_runtime"
     OR bin_exec STREQUAL "coro/test_channel"
     OR bin_exec STREQUAL "coro/test_channel_queue")
    set(heavy_thread_timeout 180)
    if(DFTRACER_UTILS_ENABLE_TSAN OR DFTRACER_UTILS_COVERAGE
       OR DFTRACER_UTILS_ENABLE_ASAN)
      set(heavy_thread_timeout 600)
    endif()
    if(DFTRACER_UTILS_ENABLE_TSAN)
      set(heavy_thread_timeout 1800)
    endif()
    set_tests_properties(${bin_exec} PROPERTIES
                                           TIMEOUT ${heavy_thread_timeout}
                                           RESOURCE_LOCK heavy_threading)
  endif()

  # Builds a dozen indexes. Each commit ends in a full fsync (F_FULLFSYNC on
  # macOS), so it waits on the disk and slows down when the rest of the suite
  # writes at the same time.
  # trace/index/test_indexer and trace/reader/test_reader build indexes too
  # and can pass their limit under the full suite.
  if(bin_exec STREQUAL "trace/index/test_build_memory"
     OR bin_exec STREQUAL "trace/index/test_indexer"
     OR bin_exec STREQUAL "trace/reader/test_reader")
    set_tests_properties(${bin_exec} PROPERTIES RUN_SERIAL TRUE)
  endif()

  # The crash test writes a few thousand file images, the concurrency test
  # commits durable batches that fsync, and the file test rewrites a damaged
  # copy of a data file a few hundred times. Each can pass its limit when the
  # suite starts them beside the rest.
  if(bin_exec STREQUAL "store/test_crash"
     OR bin_exec STREQUAL "store/test_concurrent"
     OR bin_exec STREQUAL "store/test_file")
    set_tests_properties(${bin_exec} PROPERTIES RUN_SERIAL TRUE)
  endif()

  # Eight forked writers, each with an I/O pool of one thread per hardware
  # thread, can pass the per-user thread limit when other tests run beside them.
  if(bin_exec STREQUAL "store/test_processes")
    set_tests_properties(${bin_exec} PROPERTIES RUN_SERIAL TRUE)
  endif()

  # test_big_member_reads inflates several large gzip members and can pass the
  # default limit on a small CI runner under the full suite. test_catalog and
  # test_build_memory can too, and more so under coverage.
  if(bin_exec STREQUAL "trace/index/test_big_member_reads"
     OR bin_exec STREQUAL "trace/index/test_catalog"
     OR bin_exec STREQUAL "trace/index/test_build_memory")
    set(slow_serial_timeout 300)
    if(DFTRACER_UTILS_COVERAGE)
      set(slow_serial_timeout 900)
    endif()
    if(DFTRACER_UTILS_ENABLE_TSAN)
      set(slow_serial_timeout 1800)
    endif()
    set_tests_properties(${bin_exec} PROPERTIES TIMEOUT ${slow_serial_timeout}
                                                RUN_SERIAL TRUE)
  endif()

  # The allocator counts its allocations only in debug mode, which is fixed
  # before the first allocation.
  if(bin_exec STREQUAL "trace/index/test_parse_allocations")
    set_tests_properties(${bin_exec} PROPERTIES ENVIRONMENT
                                           "DFTRACER_UTILS_MEMORY_DEBUG=1")
  endif()

  if(bin_exec STREQUAL "trace/index/test_input_syscalls")
    if(DFTRACER_UTILS_ENABLE_ASAN OR DFTRACER_UTILS_ENABLE_TSAN
       OR DFTRACER_UTILS_ENABLE_UBSAN)
      # strace and the interposer cannot run a sanitizer binary's children
      set_tests_properties(${bin_exec} PROPERTIES DISABLED TRUE)
    endif()
    add_dependencies(${target_name} input_syscalls_interpose
      dftracer_index dftracer_event_count dftracer_info dftracer_stats
      dftracer_view dftracer_split dftracer_validate dftracer_comparator
      dftracer_replay)
    target_compile_definitions(${target_name} PRIVATE
      DFTRACER_UTILS_TEST_INTERPOSE_PATH="$<TARGET_FILE:input_syscalls_interpose>"
      DFTRACER_UTILS_TEST_BIN_DIR="$<TARGET_FILE_DIR:dftracer_index>")
    set_tests_properties(${bin_exec} PROPERTIES TIMEOUT 300 RUN_SERIAL TRUE)
  endif()

  if(bin_exec STREQUAL "comm/test_observe")
    add_dependencies(${target_name} dftracer_event_count)
    set_tests_properties(${bin_exec} PROPERTIES ENVIRONMENT
      "DFTRACER_UTILS_EVENT_COUNT_PATH=$<TARGET_FILE:dftracer_event_count>")
  endif()

  if(bin_exec STREQUAL "trace/index/test_index_skip")
    add_dependencies(${target_name} dftracer_index)
    target_compile_definitions(${target_name} PRIVATE
      DFTRACER_UTILS_TEST_INDEX_BINARY="$<TARGET_FILE:dftracer_index>")
  endif()

  # Robustness test generates large (128 MB) gzip files. Coverage
  # instrumentation makes it significantly slower, and TSan more so. The prune
  # oracle reads its corpus once per query and time range.
  # The prefilter fuzz scans 300 random queries four ways.
  if(bin_exec STREQUAL "trace/reader/test_reader_robustness"
     OR bin_exec STREQUAL "trace/index/test_prune_oracle"
     OR bin_exec STREQUAL "trace/index/test_prefilter_fuzz")
    set(robustness_timeout 600)
    if(DFTRACER_UTILS_COVERAGE)
      set(robustness_timeout 1800)
    endif()
    if(DFTRACER_UTILS_ENABLE_TSAN)
      set(robustness_timeout 1800)
    endif()
    set_tests_properties(${bin_exec} PROPERTIES TIMEOUT ${robustness_timeout})
  endif()
endforeach()

# +++++++++++++++++++++++++++++++++++++++++
# Example plugins: `dftracer_plugin check` on each one examples/plugins/
# builds, so every kind the SDK offers - fold, udf, aggregate, format, sink,
# node, provider - is built and run by the default `tests` preset. A plugin
# whose golden test folder exists under examples/plugins/tests/<name> is
# checked against it; one without still has to load, print the kinds it
# provides and leave no leaks.
# +++++++++++++++++++++++++++++++++++++++++
if(TARGET dftracer_plugin)
  # One entry per example plugin that loads on its own. The two consumers,
  # perbatch_consumer and dur_stats_consumer, are left out: each refuses to load
  # without the producer that provides its port, and `check` takes one library.
  set(EXAMPLE_PLUGINS
      duration_report_plugin # fold, writes a report file
      duration_histogram_plugin # fold, emits named results
      event_counter_plugin # fold in C
      query_filter_plugin # fold with its own filter dir
      process_counts_plugin # fold in C over the process tree
      process_file_edges_plugin # fold in C over file edges
      process_collections_plugin # fold in C over collections
      process_pid_argstats_plugin # fold in C over per-pid arguments
      process_topk_plugin # fold in C over the top events
      process_regr_stats_plugin # fold in C over regression statistics
      perbatch_producer_plugin # fold in C, provides a port
      dur_stats_producer_plugin # fold in C, provides a stats port
      duration_udfs_plugin # udf
      stats_udafs_plugin # aggregate
      csv_sink_plugin # sink
      csv_reader_plugin # format
      sessionize_plugin # node
      range_provider_plugin) # provider
  foreach(plugin IN LISTS EXAMPLE_PLUGINS)
    if(NOT TARGET ${plugin})
      continue()
    endif()
    string(REPLACE "_plugin" "" example "${plugin}")
    set(golden_args "")
    set(golden_dir "${CMAKE_SOURCE_DIR}/examples/plugins/tests/${example}")
    if(EXISTS "${golden_dir}")
      list(APPEND golden_args --tests "${golden_dir}")
    endif()
    add_test(
      NAME examples/plugins/${example}
      COMMAND $<TARGET_FILE:dftracer_plugin> check $<TARGET_FILE:${plugin}>
              ${golden_args})
    set_tests_properties(examples/plugins/${example} PROPERTIES TIMEOUT 300)
  endforeach()
endif()

# +++++++++++++++++++++++++++++++++++++++++
# Global operator new
# +++++++++++++++++++++++++++++++++++++++++

# Every C++ test executable runs on our heap, as the CLI tools do. The memory
# tests count blocks and bytes exactly, so they keep the C library's operator
# new; memory_test_global_new links the replacement itself.
function(dftu_link_global_new_to_tests)
  if(NOT TARGET dftracer_utils_global_new)
    return()
  endif()
  set(pending ${CMAKE_SOURCE_DIR}/tests)
  set(targets "")
  while(pending)
    list(POP_FRONT pending dir)
    get_property(
      dir_targets
      DIRECTORY ${dir}
      PROPERTY BUILDSYSTEM_TARGETS)
    list(APPEND targets ${dir_targets})
    get_property(
      subdirs
      DIRECTORY ${dir}
      PROPERTY SUBDIRECTORIES)
    list(APPEND pending ${subdirs})
  endwhile()
  foreach(target ${targets})
    get_target_property(type ${target} TYPE)
    if(NOT type STREQUAL "EXECUTABLE" OR target MATCHES "^memory_")
      continue()
    endif()
    get_target_property(sources ${target} SOURCES)
    set(has_cxx FALSE)
    foreach(source ${sources})
      if(source MATCHES "\\.(cpp|cc|cxx)$")
        set(has_cxx TRUE)
      endif()
    endforeach()
    if(has_cxx)
      target_link_libraries(${target} PRIVATE dftracer_utils_global_new)
    endif()
  endforeach()
endfunction()

cmake_language(DEFER DIRECTORY ${CMAKE_SOURCE_DIR} CALL
               dftu_link_global_new_to_tests)
